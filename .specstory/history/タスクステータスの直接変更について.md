+     {
+         $this->authorize('update', $todo);
+ 
+         $validated = $request->validate([
+             'status' => 'required|string|in:pending,in_progress,completed'
+         ]);
+ 
+         $todo->update(['status' => $validated['status']]);

+         return response()->json(['status' => $todo->status]);
